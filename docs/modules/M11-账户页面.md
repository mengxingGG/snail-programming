## useAuth Hook

```typescript
export function useAuth() {
  const [userId, setUserId] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  // 初始化：尝试获取已保存的 session
  useEffect(() => {
    snailAPI.auth.getSession()
      .then(s => { if (s?.userId) setUserId(s.userId); })
      .catch(() => {})
      .finally(() => setIsLoading(false));
  }, []);

  const login = async (username: string, password: string) => {
    const res = await snailAPI.auth.login(username, password);
    setUserId(res.userId);
  };

  const logout = async () => {
    await snailAPI.auth.logout();
    setUserId(null);
  };

  const register = async (username: string, password: string) => {
    await snailAPI.auth.register(username, password);
    // 注册后自动登录
    await login(username, password);
  };

  return { userId, isLoading, login, logout, register, isLoggedIn: !!userId };
}
```

## 表单验证规则

| 字段 | 规则 | 错误信息 |
|------|------|---------|
| 用户名 | 不能为空 | "请输入用户名" |
| 用户名 | 2-20 字符 | "用户名2-20字符" |
| 密码 | 不能为空 | "请输入密码" |
| 密码 | ≥6 字符 | "密码至少6位" |
| 注册 | 用户名不可重复 | 后端返回 "用户名已存在" |

## 错误处理

| 错误场景 | 处理方式 |
|---------|---------|
| 登录失败 | 红色错误文字显示在表单内 |
| 注册失败（用户名重复） | 同上 |
| 密码错误 | 同上，不清空密码框 |
| 网络异常 | "网络连接失败，请检查网络" |
